--------------------------------------------------------------------------------
-- Import IEEE library
--------------------------------------------------------------------------------
LIBRARY ieee;

--------------------------------------------------------------------------------
-- Import std_logic and std_logic_vector definitions
--------------------------------------------------------------------------------
USE ieee.std_logic_1164.ALL;

--------------------------------------------------------------------------------
-- Testbench Entity
-- No ports required because test signals are generated internally
--------------------------------------------------------------------------------
ENTITY sipo_test IS
END sipo_test;

--------------------------------------------------------------------------------
-- Testbench Architecture
--------------------------------------------------------------------------------
ARCHITECTURE behavior OF sipo_test IS

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Declares the SIPO register being tested
    ----------------------------------------------------------------------------
    COMPONENT sipo
    PORT(
         data : IN  std_logic;
         clk : IN  std_logic;
         reset : IN  std_logic;
         Q : OUT std_logic_vector(3 downto 0)
        );
    END COMPONENT;

    ----------------------------------------------------------------------------
    -- Input Signals
    ----------------------------------------------------------------------------
    signal data : std_logic := '0';

    signal clk : std_logic := '0';

    signal reset : std_logic := '0';

    ----------------------------------------------------------------------------
    -- Output Signal
    ----------------------------------------------------------------------------
    signal Q : std_logic_vector(3 downto 0);

    ----------------------------------------------------------------------------
    -- Clock Period
    ----------------------------------------------------------------------------
    constant clk_period : time := 2 ps;

BEGIN

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    ----------------------------------------------------------------------------
    uut: sipo PORT MAP (

          data => data,

          clk => clk,

          reset => reset,

          Q => Q

        );

    ----------------------------------------------------------------------------
    -- Clock Generation Process
    -- Produces a continuous clock
    ----------------------------------------------------------------------------
    clk_process : process
    begin

        clk <= '0';
        wait for clk_period/2;

        clk <= '1';
        wait for clk_period/2;

    end process;

    ----------------------------------------------------------------------------
    -- Stimulus Process
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Apply Reset
        -- Clears all flip-flops
        ------------------------------------------------------------------------
        reset <= '1';

        wait for clk_period;

        ------------------------------------------------------------------------
        -- Release Reset
        ------------------------------------------------------------------------
        reset <= '0';

        wait for clk_period;

        ------------------------------------------------------------------------
        -- Apply Serial Data = 1
        ------------------------------------------------------------------------
        data <= '1';

        ------------------------------------------------------------------------
        -- Wait 4 clock cycles
        -- Bit shifts through all four stages
        ------------------------------------------------------------------------
        wait for clk_period*4;

        ------------------------------------------------------------------------
        -- GOOD PRACTICE:
        -- Stop simulation here
        ------------------------------------------------------------------------
        wait;

    end process;

END;

--------------------------------------------------------------------------------
data <= '1';
wait for clk_period;

data <= '0';
wait for clk_period;

data <= '1';
wait for clk_period;

data <= '0';
wait for clk_period;
