----------------------------------------------------------------------------------
-- Import IEEE library
----------------------------------------------------------------------------------
library IEEE;

----------------------------------------------------------------------------------
-- Import std_logic and std_logic_vector definitions
----------------------------------------------------------------------------------
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Entity Declaration
-- 4-bit Parallel In Serial Out Shift Register
----------------------------------------------------------------------------------
entity piso is

    Port (

           -- Common clock
           clk : in STD_LOGIC;

           -- Reset signal
           reset : in STD_LOGIC;

           -- Load control
           -- load=1 → Parallel Load
           -- load=0 → Shift Mode
           load : in STD_LOGIC;

           -- 4-bit parallel input
           pin : in STD_LOGIC_VECTOR (3 downto 0);

           -- Serial output
           sout : out STD_LOGIC

         );

end piso;

----------------------------------------------------------------------------------
-- Architecture Section
----------------------------------------------------------------------------------
architecture Behavioral of piso is

    ----------------------------------------------------------------------------
    -- D Flip-Flop Component
    -- Used to construct the shift register
    ----------------------------------------------------------------------------
    component d_flip_flop is

        Port (

               d : in STD_LOGIC;

               reset : in STD_LOGIC;

               clk : in STD_LOGIC;

               q : out STD_LOGIC;

               qbar : out STD_LOGIC

             );

    end component;

    ----------------------------------------------------------------------------
    -- Stores current contents of shift register
    ----------------------------------------------------------------------------
    signal q_internal : STD_LOGIC_VECTOR(3 downto 0);

    ----------------------------------------------------------------------------
    -- Inputs to each D Flip-Flop
    ----------------------------------------------------------------------------
    signal d0, d1, d2, d3 : STD_LOGIC;

begin

    ----------------------------------------------------------------------------
    -- Input Selection Logic
    --
    -- Acts like a 2:1 Multiplexer
    --
    -- load=1 → Parallel Input
    -- load=0 → Shift Data
    ----------------------------------------------------------------------------

    ----------------------------------------------------------------------------
    -- FF0 Input
    --
    -- Load Mode:
    -- d0 = pin(0)
    --
    -- Shift Mode:
    -- d0 gets previous FF1 value
    ----------------------------------------------------------------------------
    d0 <= pin(0) when load = '1'
          else q_internal(1);

    ----------------------------------------------------------------------------
    -- FF1 Input
    ----------------------------------------------------------------------------
    d1 <= pin(1) when load = '1'
          else q_internal(2);

    ----------------------------------------------------------------------------
    -- FF2 Input
    ----------------------------------------------------------------------------
    d2 <= pin(2) when load = '1'
          else q_internal(3);

    ----------------------------------------------------------------------------
    -- FF3 Input
    --
    -- During shifting, a zero enters the register
    ----------------------------------------------------------------------------
    d3 <= pin(3) when load = '1'
          else '0';

    ----------------------------------------------------------------------------
    -- Flip-Flop 0
    ----------------------------------------------------------------------------
    FF0 : d_flip_flop
    port map (

        clk   => clk,

        reset => reset,

        d     => d0,

        q     => q_internal(0)

    );

    ----------------------------------------------------------------------------
    -- Flip-Flop 1
    ----------------------------------------------------------------------------
    FF1 : d_flip_flop
    port map (

        clk   => clk,

        reset => reset,

        d     => d1,

        q     => q_internal(1)

    );

    ----------------------------------------------------------------------------
    -- Flip-Flop 2
    ----------------------------------------------------------------------------
    FF2 : d_flip_flop
    port map (

        clk   => clk,

        reset => reset,

        d     => d2,

        q     => q_internal(2)

    );

    ----------------------------------------------------------------------------
    -- Flip-Flop 3
    ----------------------------------------------------------------------------
    FF3 : d_flip_flop
    port map (

        clk   => clk,

        reset => reset,

        d     => d3,

        q     => q_internal(3)

    );

    ----------------------------------------------------------------------------
    -- Serial Output
    --
    -- Least Significant Stage leaves first
    ----------------------------------------------------------------------------
    sout <= q_internal(0);

end Behavioral;
