--------------------------------------------------------------------------------
-- Import IEEE library
--------------------------------------------------------------------------------
LIBRARY ieee;

--------------------------------------------------------------------------------
-- Import std_logic and std_logic_vector definitions
--------------------------------------------------------------------------------
USE ieee.std_logic_1164.ALL;

--------------------------------------------------------------------------------
-- Testbench Entity
-- No ports required because test signals are generated internally
--------------------------------------------------------------------------------
ENTITY updown_counter_test IS
END updown_counter_test;

--------------------------------------------------------------------------------
-- Testbench Architecture
--------------------------------------------------------------------------------
ARCHITECTURE behavior OF updown_counter_test IS

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Unit Under Test (UUT)
    ----------------------------------------------------------------------------
    COMPONENT updown_counter
    PORT(
         clk : IN  std_logic;
         reset : IN  std_logic;

         -- Direction control
         -- 1 = Count Up
         -- 0 = Count Down
         updown : IN  std_logic;

         -- Counter output
         q : OUT  std_logic_vector(3 downto 0)
        );
    END COMPONENT;

    ----------------------------------------------------------------------------
    -- Input Signals
    ----------------------------------------------------------------------------
    signal clk : std_logic := '0';

    signal reset : std_logic := '0';

    signal updown : std_logic := '0';

    ----------------------------------------------------------------------------
    -- Output Signal
    ----------------------------------------------------------------------------
    signal q : std_logic_vector(3 downto 0);

    ----------------------------------------------------------------------------
    -- Clock Period
    ----------------------------------------------------------------------------
    constant clk_period : time := 10 ns;

BEGIN

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    ----------------------------------------------------------------------------
    uut: updown_counter PORT MAP (

          clk => clk,

          reset => reset,

          updown => updown,

          q => q

        );

    ----------------------------------------------------------------------------
    -- Clock Generation Process
    -- Produces a continuous clock signal
    ----------------------------------------------------------------------------
    clk_process : process
    begin

        clk <= '0';
        wait for clk_period/2;

        clk <= '1';
        wait for clk_period/2;

    end process;

    ----------------------------------------------------------------------------
    -- Stimulus Process
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Apply Reset
        -- Counter should become 0000
        ------------------------------------------------------------------------
        reset <= '1';

        wait for 20 ns;

        ------------------------------------------------------------------------
        -- Release Reset
        ------------------------------------------------------------------------
        reset <= '0';

        ------------------------------------------------------------------------
        -- Enable Up Counting
        ------------------------------------------------------------------------
        updown <= '1';

        ------------------------------------------------------------------------
        -- Observe counter for 100 ns
        --
        -- Expected:
        -- 0000
        -- 0001
        -- 0010
        -- 0011
        -- ...
        ------------------------------------------------------------------------
        wait for 100 ns;

        ------------------------------------------------------------------------
        -- Enable Down Counting
        ------------------------------------------------------------------------
        updown <= '0';

        ------------------------------------------------------------------------
        -- Observe counter for another 100 ns
        --
        -- Expected:
        -- Current Count
        -- Current Count -1
        -- Current Count -2
        -- ...
        ------------------------------------------------------------------------
        wait for 100 ns;

        ------------------------------------------------------------------------
        -- Stop Simulation
        ------------------------------------------------------------------------
        wait;

    end process;

END;
