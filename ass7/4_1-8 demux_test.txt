----------------------------------------------------------------------------------
-- Testbench for 1:8 Demultiplexer
-- Used only for simulation and verification
----------------------------------------------------------------------------------

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Testbench entity
-- No ports are needed because test values are generated internally
----------------------------------------------------------------------------------
entity demux_test is
end demux_test;

----------------------------------------------------------------------------------
-- Testbench architecture
----------------------------------------------------------------------------------
architecture behavior of demux_test is

    ----------------------------------------------------------------------------
    -- Component declaration of the DEMUX being tested
    ----------------------------------------------------------------------------
    component demux
    Port (
           data : in STD_LOGIC;
           selectline : in STD_LOGIC_VECTOR (2 downto 0);
           out_data : out STD_LOGIC_VECTOR (7 downto 0)
         );
    end component;

    ----------------------------------------------------------------------------
    -- Internal signals used to drive the DEMUX inputs
    ----------------------------------------------------------------------------
    signal data : STD_LOGIC := '0';

    signal selectline : STD_LOGIC_VECTOR(2 downto 0)
                        := (others => '0');

    ----------------------------------------------------------------------------
    -- Signal used to observe DEMUX output
    ----------------------------------------------------------------------------
    signal out_data : STD_LOGIC_VECTOR(7 downto 0);

begin

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    -- Creates an instance of the DEMUX
    ----------------------------------------------------------------------------
    uut: demux
    port map (

        data => data,

        selectline => selectline,

        out_data => out_data

    );

    ----------------------------------------------------------------------------
    -- Stimulus Process
    -- Applies different test cases
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Test Case 1
        -- Input = 1
        -- Select = 000
        -- Expected Output = 00000001
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "000";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 2
        -- Input = 1
        -- Select = 001
        -- Expected Output = 00000010
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "001";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 3
        -- Input = 1
        -- Select = 010
        -- Expected Output = 00000100
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "010";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 4
        -- Input = 1
        -- Select = 011
        -- Expected Output = 00001000
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "011";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 5
        -- Input = 1
        -- Select = 100
        -- Expected Output = 00010000
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "100";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 6
        -- Input = 1
        -- Select = 101
        -- Expected Output = 00100000
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "101";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 7
        -- Input = 1
        -- Select = 110
        -- Expected Output = 01000000
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "110";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 8
        -- Input = 1
        -- Select = 111
        -- Expected Output = 10000000
        ------------------------------------------------------------------------
        data <= '1';
        selectline <= "111";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Stop simulation
        ------------------------------------------------------------------------
        wait;

    end process;

end behavior;
