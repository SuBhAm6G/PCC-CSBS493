--------------------------------------------------------------------------------
-- Testbench for 8:1 Multiplexer
-- A testbench is used only for simulation and verification.
-- It does not get synthesized into hardware.
--------------------------------------------------------------------------------

-- Import IEEE library
LIBRARY ieee;

-- Import std_logic and std_logic_vector datatypes
USE ieee.std_logic_1164.ALL;

-- Additional arithmetic library (not used here)
-- USE ieee.numeric_std.ALL;

--------------------------------------------------------------------------------
-- Testbench entity
-- Testbenches do not have input or output ports because they generate
-- their own test signals internally.
--------------------------------------------------------------------------------
ENTITY mux_test IS
END mux_test;

--------------------------------------------------------------------------------
-- Architecture containing the test logic
--------------------------------------------------------------------------------
ARCHITECTURE behavior OF mux_test IS

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Declares the Multiplexer module that will be tested.
    ----------------------------------------------------------------------------
    COMPONENT mux
    PORT(
         data : IN  std_logic_vector(7 downto 0);
         selectline : IN  std_logic_vector(2 downto 0);
         out_data : OUT std_logic
    );
    END COMPONENT;

    ----------------------------------------------------------------------------
    -- Internal signals used to drive the Multiplexer inputs.
    -- These act like wires connected to the UUT.
    ----------------------------------------------------------------------------
    signal data : std_logic_vector(7 downto 0) := (others => '0');

    signal selectline : std_logic_vector(2 downto 0) := (others => '0');

    ----------------------------------------------------------------------------
    -- Signal to capture Multiplexer output.
    ----------------------------------------------------------------------------
    signal out_data : std_logic;

BEGIN

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    -- Creates an instance of the Multiplexer.
    ----------------------------------------------------------------------------
    uut: mux PORT MAP (

          -- Connect testbench signal "data"
          -- to multiplexer input "data"
          data => data,

          -- Connect testbench signal "selectline"
          -- to multiplexer input "selectline"
          selectline => selectline,

          -- Connect multiplexer output
          -- to testbench signal "out_data"
          out_data => out_data
    );

    ----------------------------------------------------------------------------
    -- Stimulus Process
    -- Generates different test cases for simulation.
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Test Case 1
        -- selectline = 000
        -- Expected output = data(0)
        -- data = 01001101
        -- data(0) = 1
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "000";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 2
        -- selectline = 001
        -- Expected output = data(1)
        -- data(1) = 0
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "001";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 3
        -- selectline = 010
        -- Expected output = data(2)
        -- data(2) = 1
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "010";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 4
        -- selectline = 011
        -- Expected output = data(3)
        -- data(3) = 1
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "011";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 5
        -- selectline = 100
        -- Expected output = data(4)
        -- data(4) = 0
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "100";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 6
        -- selectline = 101
        -- Expected output = data(5)
        -- data(5) = 0
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "101";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 7
        -- selectline = 110
        -- Expected output = data(6)
        -- data(6) = 1
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "110";
        wait for 1 ps;

        ------------------------------------------------------------------------
        -- Test Case 8
        -- selectline = 111
        -- Expected output = data(7)
        -- data(7) = 0
        ------------------------------------------------------------------------
        data <= "01001101";
        selectline <= "111";
        wait for 1 ps;

    end process;

END;
