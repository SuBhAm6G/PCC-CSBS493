----------------------------------------------------------------------------------
-- Import IEEE library
----------------------------------------------------------------------------------
library IEEE;

----------------------------------------------------------------------------------
-- Import std_logic datatype definitions
----------------------------------------------------------------------------------
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Entity Declaration
-- Defines external inputs and outputs of the SR Flip-Flop
----------------------------------------------------------------------------------
entity SR is
    Port (

        -- Set input
        S : in STD_LOGIC;

        -- Reset input
        R : in STD_LOGIC;

        -- Clock input
        clk : in STD_LOGIC;

        -- Main output
        Q : out STD_LOGIC;

        -- Complement of Q
        Qbar : out STD_LOGIC

    );
end SR;

----------------------------------------------------------------------------------
-- Architecture Section
-- Contains the internal behavior of the flip-flop
----------------------------------------------------------------------------------
architecture Behavioral of SR is

    ----------------------------------------------------------------------------
    -- Internal signal used to store the current state
    -- This acts as the memory element
    ----------------------------------------------------------------------------
    signal temp : STD_LOGIC := '0';

begin

    ----------------------------------------------------------------------------
    -- Process executes whenever clock changes
    ----------------------------------------------------------------------------
    process(clk)

    begin

        ------------------------------------------------------------------------
        -- Check for rising edge of clock
        -- Flip-flop changes state only here
        ------------------------------------------------------------------------
        if rising_edge(clk) then

            --------------------------------------------------------------------
            -- S=0, R=0
            -- Hold previous value
            --------------------------------------------------------------------
            if (S='0' and R='0') then

                temp <= temp;

            --------------------------------------------------------------------
            -- S=0, R=1
            -- Reset output
            --------------------------------------------------------------------
            elsif (S='0' and R='1') then

                temp <= '0';

            --------------------------------------------------------------------
            -- S=1, R=0
            -- Set output
            --------------------------------------------------------------------
            elsif (S='1' and R='0') then

                temp <= '1';

            --------------------------------------------------------------------
            -- S=1, R=1
            -- Invalid condition
            -- Most lab implementations force output to unknown
            --------------------------------------------------------------------
            else

                temp <= 'X';

            end if;

        end if;

    end process;

    ----------------------------------------------------------------------------
    -- Assign internal state to output Q
    ----------------------------------------------------------------------------
    Q <= temp;

    ----------------------------------------------------------------------------
    -- Qbar is always opposite of Q
    ----------------------------------------------------------------------------
    Qbar <= not temp;

end Behavioral;
