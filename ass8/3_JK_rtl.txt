----------------------------------------------------------------------------------
-- Import IEEE library
----------------------------------------------------------------------------------
library IEEE;

----------------------------------------------------------------------------------
-- Import std_logic definitions
----------------------------------------------------------------------------------
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Entity Declaration
-- Defines external inputs and outputs of JK Flip-Flop
----------------------------------------------------------------------------------
entity jk is
    Port (

        -- J input
        J : in STD_LOGIC;

        -- K input
        K : in STD_LOGIC;

        -- Clock input
        clk : in STD_LOGIC;

        -- Asynchronous reset input
        reset : in STD_LOGIC;

        -- Main output
        Q : out STD_LOGIC;

        -- Complement output
        Qbar : out STD_LOGIC

    );
end jk;

----------------------------------------------------------------------------------
-- Architecture Section
----------------------------------------------------------------------------------
architecture Behavioral of jk is

    ----------------------------------------------------------------------------
    -- Internal signal used to store flip-flop state
    ----------------------------------------------------------------------------
    signal temp : STD_LOGIC := '0';

begin

    ----------------------------------------------------------------------------
    -- Process reacts to both reset and clock
    ----------------------------------------------------------------------------
    process(clk, reset)

    begin

        ------------------------------------------------------------------------
        -- Asynchronous Reset
        -- Executes immediately when reset becomes 1
        ------------------------------------------------------------------------
        if (reset = '1') then

            temp <= '0';

        ------------------------------------------------------------------------
        -- Rising edge clock detection
        ------------------------------------------------------------------------
        elsif rising_edge(clk) then

            --------------------------------------------------------------------
            -- J=0 K=0
            -- Hold previous value
            --------------------------------------------------------------------
            if (J='0' and K='0') then

                temp <= temp;

            --------------------------------------------------------------------
            -- J=0 K=1
            -- Reset
            --------------------------------------------------------------------
            elsif (J='0' and K='1') then

                temp <= '0';

            --------------------------------------------------------------------
            -- J=1 K=0
            -- Set
            --------------------------------------------------------------------
            elsif (J='1' and K='0') then

                temp <= '1';

            --------------------------------------------------------------------
            -- J=1 K=1
            -- Toggle current state
            --------------------------------------------------------------------
            else

                temp <= not temp;

            end if;

        end if;

    end process;

    ----------------------------------------------------------------------------
    -- Assign internal state to output
    ----------------------------------------------------------------------------
    Q <= temp;

    ----------------------------------------------------------------------------
    -- Complement output
    ----------------------------------------------------------------------------
    Qbar <= not temp;

end Behavioral;
