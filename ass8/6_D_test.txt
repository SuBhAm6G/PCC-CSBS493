----------------------------------------------------------------------------------
-- Import IEEE library
----------------------------------------------------------------------------------
library IEEE;

----------------------------------------------------------------------------------
-- Import std_logic definitions
----------------------------------------------------------------------------------
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Testbench Entity
-- No ports required because test values are generated internally
----------------------------------------------------------------------------------
entity dflipflop_test is
end dflipflop_test;

----------------------------------------------------------------------------------
-- Testbench Architecture
----------------------------------------------------------------------------------
architecture behavior of dflipflop_test is

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Declares the D Flip-Flop being tested
    ----------------------------------------------------------------------------
    component D
    Port (
           D : in STD_LOGIC;
           clk : in STD_LOGIC;
           Q : out STD_LOGIC;
           Qbar : out STD_LOGIC
         );
    end component;

    ----------------------------------------------------------------------------
    -- Internal test signals
    ----------------------------------------------------------------------------
    signal D_in : STD_LOGIC := '0';

    signal clk : STD_LOGIC := '0';

    ----------------------------------------------------------------------------
    -- Signals used to observe outputs
    ----------------------------------------------------------------------------
    signal Q : STD_LOGIC;
    signal Qbar : STD_LOGIC;

    ----------------------------------------------------------------------------
    -- Clock period definition
    ----------------------------------------------------------------------------
    constant clk_period : time := 1 ns;

begin

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    ----------------------------------------------------------------------------
    uut : D
    port map (

        D => D_in,

        clk => clk,

        Q => Q,

        Qbar => Qbar

    );

    ----------------------------------------------------------------------------
    -- Clock Generation Process
    -- Produces a repeating square wave
    ----------------------------------------------------------------------------
    clk_process : process
    begin

        clk <= '0';
        wait for clk_period/2;

        clk <= '1';
        wait for clk_period/2;

    end process;

    ----------------------------------------------------------------------------
    -- Stimulus Process
    -- Applies different D values
    ----------------------------------------------------------------------------
    stim_proc : process
    begin

        ------------------------------------------------------------------------
        -- Test Case 1
        -- D = 0
        -- After rising edge Q should become 0
        ------------------------------------------------------------------------
        D_in <= '0';
        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 2
        -- D = 1
        -- After rising edge Q should become 1
        ------------------------------------------------------------------------
        D_in <= '1';
        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 3
        -- D = 0
        -- After rising edge Q should become 0
        ------------------------------------------------------------------------
        D_in <= '0';
        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 4
        -- D = 1
        -- After rising edge Q should become 1
        ------------------------------------------------------------------------
        D_in <= '1';
        wait for 1 ns;

        ------------------------------------------------------------------------
        -- GOOD PRACTICE
        -- Stop the stimulus process
        ------------------------------------------------------------------------
        wait;

    end process;

end behavior;
