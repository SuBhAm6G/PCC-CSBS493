--------------------------------------------------------------------------------
-- Import IEEE library
--------------------------------------------------------------------------------
LIBRARY ieee;

--------------------------------------------------------------------------------
-- Import std_logic datatype definitions
--------------------------------------------------------------------------------
USE ieee.std_logic_1164.ALL;

--------------------------------------------------------------------------------
-- Arithmetic library not needed in this design
--------------------------------------------------------------------------------
--USE ieee.numeric_std.ALL;

--------------------------------------------------------------------------------
-- Testbench Entity
-- Testbenches do not require ports because they generate
-- their own inputs internally.
--------------------------------------------------------------------------------
ENTITY jk_test IS
END jk_test;

--------------------------------------------------------------------------------
-- Testbench Architecture
--------------------------------------------------------------------------------
ARCHITECTURE behavior OF jk_test IS

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Declares the JK Flip-Flop being tested
    ----------------------------------------------------------------------------
    COMPONENT jk
    PORT(
         j : IN  std_logic;
         k : IN  std_logic;
         Q : OUT  std_logic;
         Qbar : OUT  std_logic;
         clk : IN  std_logic;
         reset : IN  std_logic
        );
    END COMPONENT;

    ----------------------------------------------------------------------------
    -- Input signals connected to JK Flip-Flop
    ----------------------------------------------------------------------------
    signal j : std_logic := '0';
    signal k : std_logic := '0';

    ----------------------------------------------------------------------------
    -- Clock signal starts at 0
    ----------------------------------------------------------------------------
    signal clk : std_logic := '0';

    ----------------------------------------------------------------------------
    -- Reset signal starts inactive
    ----------------------------------------------------------------------------
    signal reset : std_logic := '0';

    ----------------------------------------------------------------------------
    -- Output signals used to observe results
    ----------------------------------------------------------------------------
    signal Q : std_logic;
    signal Qbar : std_logic;

    ----------------------------------------------------------------------------
    -- Clock period definition
    -- Full clock cycle = 1 ns
    ----------------------------------------------------------------------------
    constant clk_period : time := 1 ns;

BEGIN

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    -- Creates an instance of the JK Flip-Flop
    ----------------------------------------------------------------------------
    uut: jk PORT MAP (

          j => j,
          k => k,

          Q => Q,
          Qbar => Qbar,

          clk => clk,
          reset => reset

        );

    ----------------------------------------------------------------------------
    -- Clock Generation Process
    -- Generates a continuous square wave
    ----------------------------------------------------------------------------
    clk_process :process
    begin

        ------------------------------------------------------------------------
        -- Clock LOW for half cycle
        ------------------------------------------------------------------------
        clk <= '0';
        wait for clk_period/2;

        ------------------------------------------------------------------------
        -- Clock HIGH for half cycle
        ------------------------------------------------------------------------
        clk <= '1';
        wait for clk_period/2;

    end process;

    ----------------------------------------------------------------------------
    -- Stimulus Process
    -- Applies various JK input combinations
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Test Case 1
        -- J=0 K=0
        -- Hold State
        ------------------------------------------------------------------------
        j <= '0';
        k <= '0';
        reset <= '0';

        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 2
        -- J=0 K=1
        -- Reset State
        -- Expected Q = 0
        ------------------------------------------------------------------------
        j <= '0';
        k <= '1';
        reset <= '0';

        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 3
        -- J=1 K=0
        -- Set State
        -- Expected Q = 1
        ------------------------------------------------------------------------
        j <= '1';
        k <= '0';
        reset <= '0';

        wait for 1 ns;

        ------------------------------------------------------------------------
        -- Test Case 4
        -- J=1 K=1
        -- Toggle State
        -- Q changes to opposite value at clock edge
        ------------------------------------------------------------------------
        j <= '1';
        k <= '1';
        reset <= '0';

        wait for 1 ns;

        ------------------------------------------------------------------------
        -- GOOD PRACTICE:
        -- Add wait; at the end to stop the stimulus process.
        --
        -- Without wait;
        -- the process restarts from the beginning forever.
        ------------------------------------------------------------------------

        wait;

    end process;
    ..
reset <= '1';
wait for 1 ns;

reset <= '0';
wait for 1 ns;

END;
