----------------------------------------------------------------------------------
-- Testbench for SR Flip-Flop
-- Used only for simulation and verification
----------------------------------------------------------------------------------

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

----------------------------------------------------------------------------------
-- Testbench entity
-- No ports are required because test signals are generated internally
----------------------------------------------------------------------------------
entity SRtest is
end SRtest;

----------------------------------------------------------------------------------
-- Testbench Architecture
----------------------------------------------------------------------------------
architecture behavior of SRtest is

    ----------------------------------------------------------------------------
    -- Component Declaration
    -- Declares the SR Flip-Flop to be tested
    ----------------------------------------------------------------------------
    component SR
    Port(
         S : in STD_LOGIC;
         R : in STD_LOGIC;
         clk : in STD_LOGIC;
         Q : out STD_LOGIC;
         Qbar : out STD_LOGIC
    );
    end component;

    ----------------------------------------------------------------------------
    -- Internal signals connected to the SR Flip-Flop inputs
    ----------------------------------------------------------------------------
    signal S : STD_LOGIC := '0';
    signal R : STD_LOGIC := '0';
    signal clk : STD_LOGIC := '0';

    ----------------------------------------------------------------------------
    -- Signals used to observe outputs
    ----------------------------------------------------------------------------
    signal Q : STD_LOGIC;
    signal Qbar : STD_LOGIC;

begin

    ----------------------------------------------------------------------------
    -- UUT = Unit Under Test
    -- Instantiate the SR Flip-Flop
    ----------------------------------------------------------------------------
    uut: SR
    port map(
        S => S,
        R => R,
        clk => clk,
        Q => Q,
        Qbar => Qbar
    );

    ----------------------------------------------------------------------------
    -- Clock Generation Process
    -- Creates a repeating clock signal
    ----------------------------------------------------------------------------
    clk_process : process
    begin

        clk <= '0';
        wait for 5 ns;

        clk <= '1';
        wait for 5 ns;

    end process;

    ----------------------------------------------------------------------------
    -- Stimulus Process
    -- Applies different input combinations
    ----------------------------------------------------------------------------
    stim_proc: process
    begin

        ------------------------------------------------------------------------
        -- Test Case 1 : Hold State
        -- S=0 R=0
        -- Output should retain previous value
        ------------------------------------------------------------------------
        S <= '0';
        R <= '0';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Test Case 2 : Set State
        -- S=1 R=0
        -- Q becomes 1
        ------------------------------------------------------------------------
        S <= '1';
        R <= '0';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Test Case 3 : Hold State Again
        -- S=0 R=0
        -- Q should remain 1
        ------------------------------------------------------------------------
        S <= '0';
        R <= '0';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Test Case 4 : Reset State
        -- S=0 R=1
        -- Q becomes 0
        ------------------------------------------------------------------------
        S <= '0';
        R <= '1';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Test Case 5 : Hold State Again
        -- Q should remain 0
        ------------------------------------------------------------------------
        S <= '0';
        R <= '0';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Test Case 6 : Invalid State
        -- S=1 R=1
        -- Output becomes invalid/unknown
        ------------------------------------------------------------------------
        S <= '1';
        R <= '1';
        wait for 10 ns;

        ------------------------------------------------------------------------
        -- Stop Simulation
        ------------------------------------------------------------------------
        wait;

    end process;

end behavior;
